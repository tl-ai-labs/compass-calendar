  it("renders no attendee badge when an event has no attendees", () => {
    const { unmount: unmountTimedUndefined } = render(
      <TimedEventCard
        displayMode="saved"
        event={createEvent({ attendees: undefined })}
        motionMode="idle"
        position={position}
      />,
    );
    expect(screen.queryByTestId("event-attendee-badge")).toBeNull();
    unmountTimedUndefined();

    const { unmount: unmountTimedEmpty } = render(
      <TimedEventCard
        displayMode="saved"
        event={createEvent({ attendees: [] })}
        motionMode="idle"
        position={position}
      />,
    );
    expect(screen.queryByTestId("event-attendee-badge")).toBeNull();
    unmountTimedEmpty();

    const { unmount: unmountAllDayUndefined } = render(
      <AllDayEventCard
        event={createEvent({ isAllDay: true, attendees: undefined })}
        focusColor="#3b82f6"
        isPlaceholder={false}
        position={position}
      />,
    );
    expect(screen.queryByTestId("event-attendee-badge")).toBeNull();
    unmountAllDayUndefined();

    const { unmount: unmountAllDayEmpty } = render(
      <AllDayEventCard
        event={createEvent({ isAllDay: true, attendees: [] })}
        focusColor="#3b82f6"
        isPlaceholder={false}
        position={position}
      />,
    );
    expect(screen.queryByTestId("event-attendee-badge")).toBeNull();
    unmountAllDayEmpty();
  });

        })}

  });

  it("keeps the timed title line clamp unchanged when an event has no attendees", () => {
    const longTitle =
      "Journaling-Flow-Experiment with James: Part 2/2: The Miner's Sifting Pan";

    const { unmount } = render(
      <TimedEventCard
        displayMode="saved"
        event={createEvent({
          startDate: "2099-01-15T09:00:00.000Z",
          endDate: "2099-01-15T10:00:00.000Z",
          title: longTitle,
        })}
        motionMode="idle"
        position={position}
      />,
    );

    const titleWithoutAttendees = screen.getByText(longTitle);
    expect(titleWithoutAttendees.style.webkitLineClamp).toBe("3");
    unmount();

    // The badge takes one 16px row out of the clamp budget: the 47px the time
    // label leaves becomes 31px, which getLineClamp turns from 3 lines into 2.
    render(
      <TimedEventCard
        displayMode="saved"
        event={createEvent({
          startDate: "2099-01-15T09:00:00.000Z",
          endDate: "2099-01-15T10:00:00.000Z",
          title: longTitle,
          attendees: [
            {
              displayName: "Ada",
              email: "ada@example.com",
              responseStatus: "accepted",
            },
            {
              displayName: "Bo",
              email: "bo@example.com",
              responseStatus: "needsAction",
            },
          ],
        })}
        motionMode="idle"
        position={position}
      />,
    );

    const titleWithAttendees = screen.getByText(longTitle);
    expect(titleWithAttendees.style.webkitLineClamp).toBe("2");
  });
